# Créer un user
curl -s -X POST http://localhost:8080/api/users \
  -H "Content-Type: application/json" \
  -d '{"username":"DemoUser","email":"demo@test.com","discordId":"999"}' | jq


# Créer la guilde (owner = objet avec id)
curl -s -X POST http://localhost:8080/api/guilds \
  -H "Content-Type: application/json" \
  -d '{
    "name": "DemoGuild",
    "description": "Guilde de démonstration",
    "owner": {"id": 1},
    "memberLimit": 100
  }' | jq



curl -s -X POST http://localhost:8080/api/messages \
  -H "Content-Type: application/json" \
  -d '{
    "content": "Hello démo!",
    "author": {"id": 1},
    "channel": {"id": 1}
  }' | jq
