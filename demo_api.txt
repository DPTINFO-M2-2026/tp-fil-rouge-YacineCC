# Créer un user
curl -s -X POST http://localhost:8080/api/users \
  -H "Content-Type: application/json" \
  -d '{"username":"DemoUser","discriminator":"0001","email":"demo@test.com","discordId":"999"}' | jq


# Créer la guilde
curl -s -X POST http://localhost:8080/api/guilds \
  -H "Content-Type: application/json" \
  -d '{
    "name": "DemoGuild",
    "description": "Guilde de démonstration",
    "ownerId": 1,
    "memberLimit": 100
  }' | jq
